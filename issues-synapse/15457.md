```diff
-            txn.execute("SELECT COUNT(*) FROM " + TEMP_TABLE + "_rooms")
-            result = txn.fetchone()
-            assert result is not None
-            progress["remaining"] = result[0]
+            #txn.execute("SELECT COUNT(*) FROM " + TEMP_TABLE + "_rooms")
+            #result = txn.fetchone()
+            #assert result is not None
+            #progress["remaining"] = result[0]
```
